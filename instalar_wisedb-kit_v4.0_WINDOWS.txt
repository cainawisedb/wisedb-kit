New-Item -ItemType Directory -Force -Path $Destino | Out-Null
$sha = [System.Security.Cryptography.SHA256]::Create(); $erros = 0
foreach ($nome in $K.Keys) {
  try {
    $gz = [Convert]::FromBase64String(($K[$nome][1] -replace '\s', ''))
    $in = New-Object System.IO.MemoryStream(,$gz)
    $zs = New-Object System.IO.Compression.GZipStream($in, [System.IO.Compression.CompressionMode]::Decompress)
    $out = New-Object System.IO.MemoryStream
    $b = New-Object byte[] 65536
    while (($r = $zs.Read($b, 0, $b.Length)) -gt 0) { $out.Write($b, 0, $r) }
    $zs.Close(); $bytes = $out.ToArray()
    $h = ([BitConverter]::ToString($sha.ComputeHash($bytes)) -replace '-', '').ToLower()
    if ($h -ne $K[$nome][0]) { Write-Host "ERRO: $nome corrompido na colagem (hash diferente)" -ForegroundColor Red; $erros++; continue }
    [System.IO.File]::WriteAllBytes((Join-Path $Destino $nome), $bytes)
    Write-Host ("OK  {0,-36} {1,8} bytes" -f $nome, $bytes.Length)
  } catch { Write-Host "ERRO ao gravar $nome : $($_.Exception.Message)" -ForegroundColor Red; $erros++ }
}
if ($erros -eq 0) {
  Write-Host ""; Write-Host "OK: $($K.Count) arquivos em $Destino. Para coletar:" -ForegroundColor Green
  Write-Host "  cd $Destino; powershell -ExecutionPolicy Bypass -File .\wisedb_coleta_auto.ps1"
} else { Write-Host "ERRO: $erros arquivo(s) com problema. Cole o instalador novamente." -ForegroundColor Red; exit 1 }
